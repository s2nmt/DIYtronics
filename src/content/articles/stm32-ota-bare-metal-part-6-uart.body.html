Source: bootloader/lib/drivers/inc/uart.h, bootloader/lib/drivers/src/uart.c.

USART1 is the OTA data pipe and the log console. Pins PA9/PA10 were already set in Part 5’s GPIO_Init(). This file only programs the USART peripheral: baud, RX interrupt into a ring buffer, and polled TX.

1. Registers and masks

USART1 is mapped at 0x40013800. The header exposes SR, DR, BRR, CR1, … The driver writes control/status with full-register masks (not bitfield members), as noted in the header:

#define USART1 ((USART_TypeDef*)0x40013800)

#define USART_SR_RXNE   (1UL << 5)   /* RX data ready */
#define USART_SR_TXE    (1UL << 7)   /* TX data register empty */
#define USART_SR_ORE    (1UL << 3)   /* overrun */

#define USART_CR1_RE     (1UL << 2)  /* receiver enable */
#define USART_CR1_TE     (1UL << 3)  /* transmitter enable */
#define USART_CR1_RXNEIE (1UL << 5)  /* RXNE interrupt enable */
#define USART_CR1_UE     (1UL << 13) /* USART enable */

#define UART_BAUD_115200 115200U

OTA and logs use 115200 baud. Match that on the PC (main.py --baud).

2. RX ring buffer state

#define UART1_RX_BUF_SIZE 512U

static volatile uint8_t  s_rx_buf[UART1_RX_BUF_SIZE];
static volatile uint16_t s_rx_head;   /* IRQ writes here */
static volatile uint16_t s_rx_tail;   /* uart1_rx_get reads here */

512 bytes is enough to absorb UART bursts while main / OTA is busy erasing Flash or verifying. head == tail means empty. If the buffer is full, the IRQ drops new bytes rather than overwriting unread data.

3. uart_brr_calc — baud rate register

static uint16_t uart_brr_calc(uint32_t pclk_hz, uint32_t baud)
{
    uint32_t divider = (25U * pclk_hz) / (4U * baud);
    uint32_t mantissa = divider / 100U;
    uint32_t fraction = (((divider - mantissa * 100U) * 16U) + 50U) / 100U;

    if (fraction >= 16U) {
        fraction = 0U;
        mantissa++;
    }
    return (uint16_t)((mantissa << 4) | (fraction & 0xFU));
}

STM32 USART baud is programmed as mantissa + fraction/16 in USART_BRR. This helper follows the reference-manual style formula using the USART clock (SystemCoreClock after Part 4’s PLL setup — on this wiring USART1 runs from that PCLK). Wrong SystemCoreClock → wrong baud → garbage on the wire.

4. Uart1_Init()

void Uart1_Init(void)
{
    enable_RCC_APB2ENR(APB2_USART1);
    USART1->USART_CR1.REG = 0U;
    USART1->USART_CR2.REG = 0U;
    USART1->USART_CR3.REG = 0U;
    USART1->USART_BRR.REG = uart_brr_calc(SystemCoreClock, UART_BAUD_115200);

    s_rx_head = 0U;
    s_rx_tail = 0U;

    /* UE | TE | RE | RXNEIE */
    USART1->USART_CR1.REG =
        USART_CR1_UE | USART_CR1_TE | USART_CR1_RE | USART_CR1_RXNEIE;
    nvic_irq_enable(USART1_IRQn);
}
  1. Clock — enable USART1 on APB2 (pins were already clocked/configured in GPIO_Init).
  2. Reset CR1/CR2/CR3 — known clean state (8N1, no fancy modes).
  3. BRR — 115200 from current SystemCoreClock.
  4. Empty the software ring — head/tail = 0.
  5. Enable USART — UE (peripheral on), TE (TX), RE (RX), RXNEIE (interrupt when a byte arrives).
  6. NVIC — unmask USART1_IRQn so USART1_IRQHandler can run.

5. USART1_IRQHandler — push into the ring

void USART1_IRQHandler(void)
{
    uint32_t sr = USART1->USART_SR.REG;

    /* RXNE or ORE: reading DR clears the flag(s) */
    if ((sr & (USART_SR_RXNE | USART_SR_ORE)) != 0U) {
        uint8_t b = (uint8_t)(USART1->USART_DR.REG & 0xFFU);
        uint16_t next = (uint16_t)((s_rx_head + 1U) % UART1_RX_BUF_SIZE);

        if (next != s_rx_tail) {
            s_rx_buf[s_rx_head] = b;
            s_rx_head = next;
        }
    }
}
  • Handle RXNE (byte ready) and ORE (overrun): you must read DR to clear those flags, or the IRQ can stick.
  • Compute next = (head + 1) % 512. If next == tail, buffer is full → drop this byte.
  • Otherwise store at head and advance head.

The foreground never spins on RXNE for OTA; it only calls uart1_rx_get().

6. uart1_rx_get / uart1_rx_flush

int uart1_rx_get(uint8_t *out)
{
    if (out == NULL) return 0;

    __asm volatile("cpsid i" ::: "memory");
    if (s_rx_head == s_rx_tail) {
        __asm volatile("cpsie i" ::: "memory");
        return 0;                 /* empty */
    }
    *out = s_rx_buf[s_rx_tail];
    s_rx_tail = (uint16_t)((s_rx_tail + 1U) % UART1_RX_BUF_SIZE);
    __asm volatile("cpsie i" ::: "memory");
    return 1;                     /* got one byte */
}

cpsid i / cpsie i briefly mask IRQs so the handler cannot move head while you update tail (avoids a race on the ring indices).

void uart1_rx_flush(void)
{
    __asm volatile("cpsid i" ::: "memory");
    s_rx_head = 0U;
    s_rx_tail = 0U;
    while ((USART1->USART_SR.REG & (USART_SR_RXNE | USART_SR_ORE)) != 0U) {
        (void)USART1->USART_DR.REG;   /* drain hardware too */
    }
    __asm volatile("cpsie i" ::: "memory");
}

Clears both the software ring and any pending byte still sitting in the USART data register — useful before starting a clean OTA session.

7. TX — polled uart1_putc / uart1_write

static void uart1_putc(char c)
{
    while ((USART1->USART_SR.REG & USART_SR_TXE) == 0U) {
    }
    USART1->USART_DR.REG = (uint8_t)c;
}

void uart1_write(const char *buf, int len)
{
    for (int i = 0; i < len; i++) {
        uart1_putc(buf[i]);
    }
}

TX waits until TXE (transmit data register empty), then writes one byte. No TX IRQ — fine for logs and small ACKs. In main.c, newlib’s _write calls uart1_write, so printf(...) goes out USART1.

8. Wiring

USB–UART adapter: adapter TX → PA10 (MCU RX), adapter RX → PA9 (MCU TX), GND common. Baud 115200 8N1.
