Source: command handlers and session state in bootloader/lib/boot/src/ota.c.

Part 12 delivered a complete frame into rx_cmd / rx_payload. This post is what ota_process_frame() does with START, DATA, and END — plus early slot rejection, ACK/NACK, and the jump after a successful update.

1. Session state

static uint32_t fw_size;
static uint32_t fw_crc32;
static uint32_t fw_written;
static uint16_t fw_seq_expected;
static uint32_t fw_write_addr;
static uint32_t fw_slot_size;
static uint8_t  fw_slot_checked; /* vector/link check already done */
static uint8_t  fw_aborted;      /* early reject — NACK every frame after */

START fills these; DATA advances fw_written / sequence; END checks them against Flash. Once fw_aborted is set, DATA always fails so the host cannot keep writing after a wrong-slot image was detected.

2. Dispatch — ota_process_frame

static void ota_process_frame(void)
{
    int rc = -1;

    switch (rx_cmd) {
    case OTA_CMD_START:
        rc = ota_handle_start();
        break;
    case OTA_CMD_DATA:
        rc = ota_handle_data();
        break;
    case OTA_CMD_END:
        /* Success: handle_end ACKs + jumps; does not return. */
        rc = ota_handle_end();
        break;
    default:
        break;
    }

    if (rc == 0) {
        ota_tx_ack();
    } else {
        ota_tx_nack();
    }
}

Unknown commands NACK. On END success the handler sends its own ACK and jumps before this function would ACK again.

3. START — target inactive slot and erase

Payload is exactly 8 bytes: little-endian fw_size then little-endian expected fw_crc32 of the full image blob.

static int ota_handle_start(void)
{
    if (rx_len != 8U) {
        return -1;
    }

    fw_size  = /* LE u32 from rx_payload[0..3] */;
    fw_crc32 = /* LE u32 from rx_payload[4..7] */;

    uint32_t active   = app_get_active_start();
    uint32_t inactive = app_get_inactive_start();

    /* Only program the inactive bank — never overwrite the live app. */
    if ((inactive != APP1_START && inactive != APP2_START) ||
        (inactive == active)) {
        return -1;
    }

    fw_write_addr   = inactive;
    fw_slot_size    = ota_slot_size(fw_write_addr);
    fw_written      = 0U;
    fw_seq_expected = 0U;
    fw_slot_checked = 0U;
    fw_aborted      = 0U;

    if (fw_size == 0U || fw_size > fw_slot_size) {
        return -1;
    }

    flash_unlock();
    /* Erase only enough pages for fw_size (rounded up), not the whole slot. */
    erase_size = (fw_size + FLASH_PAGE_SIZE - 1U) & ~(FLASH_PAGE_SIZE - 1U);
    flash_erase_region(fw_write_addr, erase_size);
    …
}
  1. Parse size + image CRC32 from the START payload.
  2. Resolve inactive slot via Part 10 helpers; refuse if it looks invalid.
  3. Reset write pointer, sequence, abort flags.
  4. Reject zero / oversized images.
  5. Unlock Flash and erase a region covering the image (faster than wiping the full 20.5 KB every time).

ACK means “ready for DATA seq 0.”

4. DATA — program chunks + early slot check

static int ota_handle_data(void)
{
    uint16_t seq;
    uint32_t chunk_len;
    uint8_t  pad[OTA_CHUNK_SIZE + 2U];

    if (fw_aborted != 0U) {
        return -1;
    }

    seq = (uint16_t)rx_payload[0] | ((uint16_t)rx_payload[1] << 8);
    if (seq != fw_seq_expected) {
        return -1;
    }

    chunk_len = (uint32_t)rx_len - 2U;
    if ((fw_written + chunk_len) > fw_size || chunk_len == 0U) {
        return -1;
    }

    memcpy(pad, rx_payload + 2U, chunk_len);
    if (chunk_len & 1U) {
        pad[chunk_len++] = 0xFFU;   /* Flash half-word writes need even length */
    }

    if (flash_write_buffer(fw_write_addr + fw_written, pad, chunk_len) != 0) {
        return -1;
    }

    fw_written     += (uint32_t)rx_len - 2U;  /* count real bytes, not pad */
    fw_seq_expected = (uint16_t)(fw_seq_expected + 1U);

    if (ota_early_slot_check() != 0) {
        fw_aborted = 1U;
        flash_lock();
        return -1;
    }
    return 0;
}
  • Strict sequence numbers — out-of-order → NACK.
  • Odd-length chunks get a 0xFF pad for half-word Flash programming; fw_written still tracks the host’s byte count.
  • After enough bytes exist, ota_early_slot_check() runs.

4.1 ota_early_slot_check

/* Once header + SP/Reset exist: reject wrong-linked images before transfer ends. */
if (fw_written < sizeof(img_header_t)) return 0;

img_header_parse(fw_write_addr, &hdr);
img_payload_bounds(...);

need = (payload_addr - fw_write_addr) + 8U;  /* header + SP + Reset */
if (fw_written < need) return 0;

if (payload_addr != (fw_write_addr + IMG_HEADER_SIZE)) return -1;
if (!app_is_valid_at(payload_addr)) {
    printf("OTA: wrong slot img\r\n");
    return -1;
}
fw_slot_checked = 1U;

An APP1-linked binary written into APP2 fails here (~264 bytes in) instead of after a full transfer and crypto verify. Sets fw_aborted so later DATA frames NACK.

5. END — CRC32, verify, swap, jump

static int ota_handle_end(void)
{
    if (fw_written != fw_size) {
        printf("OTA: size mismatch\r\n");
        return -1;
    }

    calc = crc32_flash(fw_write_addr, fw_size);
    if (calc != fw_crc32) {
        printf("OTA: CRC fail\r\n");
        flash_lock();
        return -1;
    }

    if (fw_write_addr == app_get_active_start()) {
        flash_lock();
        return -1;   /* must never commit onto the active bank */
    }

    if (boot_verify_slot(fw_write_addr, slot_size, &jump_addr) != 0) {
        printf("OTA: verify fail\r\n");
        flash_lock();
        return -1;
    }

    if (jump_addr != (fw_write_addr + IMG_HEADER_SIZE)) {
        printf("OTA: bad link\r\n");
        flash_lock();
        return -1;
    }

    /* Flip active_slot to the bank we just programmed */
    cfg.magic       = OTA_MAGIC_NONE;
    cfg.active_slot = (fw_write_addr == APP2_START) ? OTA_SLOT_APP2 : OTA_SLOT_APP1;
    cfg.app_size    = fw_size;
    cfg.crc32       = fw_crc32;
    cfg.version     = 0U;

    flash_erase_page(OTA_CONFIG_ADDR);
    flash_write_buffer(OTA_CONFIG_ADDR, (const uint8_t *)&cfg, sizeof(cfg));
    flash_lock();

    printf("OTA OK %08lX\r\n", (unsigned long)jump_addr);
    ota_tx_ack();
    jump_to_application(jump_addr);
    return 0;
}
  1. Length — every START byte must have been written.
  2. CRC-32 — recompute over Flash (poly 0xEDB88320, init/final invert) and match START’s fw_crc32. Transport integrity before crypto.
  3. SHA-256 + ECDSA — boot_verify_slot (Part 11). Bad signature → leave previous active slot untouched.
  4. Link check — jump address must be slot + 0x100.
  5. Commit — erase/write OTA config with new active_slot, magic cleared.
  6. ACK + jump — print OTA OK (host tool matches this string), ACK, then jump_to_application. Does not return to main on success.

6. Failure behavior

  • Any handler failure → NACK; previous active_slot unchanged → next boot still picks the old verified image.
  • fw_aborted after early slot check → every subsequent DATA NACKs until the host stops or idle timeout returns to main.
  • Idle timeout from Part 12 returns from ota_run; main may boot_pick_slot again or re-enter OTA.
