Source: bootloader/lib/drivers/inc/flash.h, bootloader/lib/drivers/src/flash.c.

OTA must erase and program the inactive app slot while UART chunks arrive. On the F103 that means talking to the Flash controller at 0x40022000: unlock, page erase, half-word program, then lock. This driver also supplies the ACR helpers Part 4 uses before raising SYSCLK to 72 MHz.

1. Memory map and register block

/* STM32F103C8: 64 KB main Flash, 1 KB page */
#define FLASH_MEM_BASE  0x08000000UL
#define FLASH_MEM_SIZE  0x00010000UL
#define FLASH_MEM_END   (FLASH_MEM_BASE + FLASH_MEM_SIZE)
#define FLASH_PAGE_SIZE 0x00000400UL

#define FLASH_KEY1 0x45670123UL
#define FLASH_KEY2 0xCDEF89ABUL

#define FLASH ((FLASH_TypeDef *)0x40022000UL)
  • Main Flash — 0x08000000 … 0x08010000 (64 KB). Every erase/write argument is checked against this window.
  • Page size — 1 KB on medium-density F103. Erase is always whole pages.
  • KEY1 / KEY2 — the unlock sequence required by the reference manual before FLASH_CR accepts erase/program bits.

Important registers used below:

  • FLASH_ACR — latency and prefetch (clock bring-up).
  • FLASH_KEYR — write KEY1 then KEY2 to clear LOCK.
  • FLASH_SR — BSY, PGERR, WRPRTERR, EOP.
  • FLASH_CR — PG (program), PER (page erase), STRT, LOCK.
  • FLASH_AR — page address for erase.
  • FLASH_WRPR — write-protection bits (option bytes mirror).

Status/control ops use full .REG masks. Comments in the C file call out why: bitfield read-modify-write is unsafe on FLASH_CR/FLASH_SR (some bits are write-only or clear-by-write-1).

2. Latency and prefetch (clock path)

void flash_prefetch_enable(void)
{
    uint32_t acr = FLASH->FLASH_ACR.REG;
    acr |= FLASH_ACR_PRFTBE_Msk;
    FLASH->FLASH_ACR.REG = acr;
}

void flash_latency_config(uint32_t latency)
{
    uint32_t acr = FLASH->FLASH_ACR.REG;
    acr &= ~FLASH_ACR_LATENCY_Msk;
    acr |= (latency & FLASH_ACR_LATENCY_Msk);
    FLASH->FLASH_ACR.REG = acr;
}

SystemClock_Config() calls these with latency 2 before switching to 72 MHz. At that speed the Flash needs wait states; prefetch hides some of the cost. Wrong order → hard fault while fetching instructions.

3. Unlock / lock

int flash_unlock(void)
{
    if ((FLASH->FLASH_CR.REG & FLASH_CR_LOCK) != 0U) {
        FLASH->FLASH_KEYR = FLASH_KEY1;
        FLASH->FLASH_KEYR = FLASH_KEY2;
    }
    return ((FLASH->FLASH_CR.REG & FLASH_CR_LOCK) != 0U) ? -1 : 0;
}

void flash_lock(void)
{
    uint32_t timeout = FLASH_OP_TIMEOUT;
    while ((FLASH->FLASH_SR.REG & FLASH_SR_BSY) != 0U) {
        if (--timeout == 0U) {
            break;
        }
    }
    FLASH->FLASH_CR.REG |= FLASH_CR_LOCK;
}
  1. Unlock — if LOCK is set, write KEY1 then KEY2 to FLASH_KEYR. Re-read LOCK; still set → return -1.
  2. Lock — wait until BSY clears (or timeout), then set LOCK so accidental writes cannot program Flash.

OTA unlocks once at START, then locks after END (or on failure).

4. Range check and WRPR

static int flash_addr_in_range(uint32_t addr, uint32_t len)
{
    if (len == 0U) {
        return 0;
    }
    if (addr < FLASH_MEM_BASE) {
        return -1;
    }
    if ((addr + len) < addr) {
        return -1;   /* overflow */
    }
    if ((addr + len) > FLASH_MEM_END) {
        return -1;
    }
    return 0;
}

static int flash_page_is_protected(uint32_t page_addr)
{
    uint32_t page = (page_addr - FLASH_MEM_BASE) / FLASH_PAGE_SIZE;
    uint32_t wrp_bit = page / 2U;

    if (wrp_bit >= 32U) {
        return 0;
    }
    /* WRPR bit = 0: write protection active on 2-page group */
    if ((FLASH->FLASH_WRPR & (1UL << wrp_bit)) == 0U) {
        return 1;
    }
    return 0;
}
  • Range — reject anything outside the 64 KB array, and reject length overflow (addr + len wraps).
  • WRPR — on F103 each protection bit covers two 1 KB pages. Bit cleared (0) means that pair is write-protected. Erase/program refuse that page instead of relying only on the hardware WRPRTERR flag.

5. Wait for ready / done

static void flash_clear_status(void)
{
    /* RM: write 1 to clear PGERR, WRPRTERR, EOP */
    FLASH->FLASH_SR.REG = FLASH_SR_PGERR | FLASH_SR_WRPRTERR | FLASH_SR_EOP;
}

static int flash_wait_ready(void)
{
    uint32_t timeout = FLASH_OP_TIMEOUT;
    while (((FLASH->FLASH_SR.REG & FLASH_SR_BSY) != 0U) && (timeout > 0U)) {
        timeout--;
    }
    return (timeout > 0U) ? 0 : -1;
}

static int flash_wait_done(void)
{
    if (flash_wait_ready() != 0) {
        return -1;
    }
    if ((FLASH->FLASH_SR.REG & (FLASH_SR_PGERR | FLASH_SR_WRPRTERR)) != 0U) {
        flash_clear_status();
        return -1;
    }
    flash_clear_status();
    return 0;
}
  1. Spin on BSY with a software timeout (~1M loops).
  2. After an op, treat PGERR (programming error) or WRPRTERR as failure.
  3. Clear sticky status bits by writing 1 (RM0008 rule).

6. Page erase and region erase

int flash_erase_page(uint32_t page_addr)
{
    if ((page_addr & (FLASH_PAGE_SIZE - 1U)) != 0U) {
        return -1;   /* must be page-aligned */
    }
    if (flash_addr_in_range(page_addr, FLASH_PAGE_SIZE) != 0) {
        return -1;
    }
    if (flash_page_is_protected(page_addr)) {
        return -1;
    }

    flash_clear_op_bits();
    if (flash_wait_ready() != 0) {
        return -1;
    }
    flash_clear_status();

    /* PER then STRT via OR — do not RMW bitfields (STRT is write-only) */
    FLASH->FLASH_CR.REG |= FLASH_CR_PER;
    FLASH->FLASH_AR = page_addr;
    FLASH->FLASH_CR.REG |= FLASH_CR_STRT;

    if (flash_wait_done() != 0) {
        FLASH->FLASH_CR.REG &= ~FLASH_CR_PER;
        return -1;
    }
    FLASH->FLASH_CR.REG &= ~FLASH_CR_PER;
    return 0;
}
  1. Require a page-aligned address inside Flash, not WRPR-protected.
  2. Clear leftover PG/PER, wait idle, clear status.
  3. Set PER, load FLASH_AR, set STRT — hardware erases that 1 KB page.
  4. Wait done; clear PER.
int flash_erase_region(uint32_t start, uint32_t size)
{
    uint32_t addr;
    uint32_t end;

    if (size == 0U) {
        return 0;
    }
    if (flash_addr_in_range(start, size) != 0) {
        return -1;
    }

    addr = start & ~(FLASH_PAGE_SIZE - 1U);
    end = start + size;

    while (addr < end) {
        if (flash_erase_page(addr) != 0) {
            return -1;
        }
        addr += FLASH_PAGE_SIZE;
    }
    return 0;
}

OTA START rounds the image size up to whole pages and calls flash_erase_region on the inactive slot only — faster than wiping the entire slot every time.

7. Half-word program — flash_write_buffer

F1 Flash programs in 16-bit units. Address and length must both be even.

int flash_write_buffer(uint32_t addr, const uint8_t *data, uint32_t len)
{
    uint32_t i;

    if (data == NULL) {
        return -1;
    }
    if (len == 0U) {
        return 0;
    }
    if ((addr & 1U) != 0U || (len & 1U) != 0U) {
        return -1;
    }
    if (flash_addr_in_range(addr, len) != 0) {
        return -1;
    }

    for (i = 0U; i < len; i += 2U) {
        uint32_t write_addr = addr + i;
        uint16_t half = (uint16_t)data[i]
                      | ((uint16_t)data[i + 1U] << 8);
        volatile uint16_t *flash_ptr = (volatile uint16_t *)write_addr;

        if (flash_page_is_protected(write_addr & ~(FLASH_PAGE_SIZE - 1U))) {
            return -1;
        }

        flash_clear_op_bits();
        if (flash_wait_ready() != 0) {
            return -1;
        }
        flash_clear_status();

        FLASH->FLASH_CR.REG |= FLASH_CR_PG;
        *flash_ptr = half;

        if (flash_wait_done() != 0) {
            FLASH->FLASH_CR.REG &= ~FLASH_CR_PG;
            return -1;
        }
        FLASH->FLASH_CR.REG &= ~FLASH_CR_PG;

        if (*flash_ptr != half) {
            return -1;   /* verify readback */
        }
    }
    return 0;
}
  1. Pack two bytes little-endian into a half-word.
  2. Set PG, store the half-word to the Flash address (that store triggers the program pulse).
  3. Wait for done, clear PG.
  4. Read back the half-word; mismatch → fail (catches stuck bits / protect issues early).

OTA DATA pads an odd chunk with 0xFF so len stays even before calling this.

8. Reads

uint16_t flash_read_halfword(uint32_t addr)
{
    return *(volatile const uint16_t *)addr;
}

uint32_t flash_read_word(uint32_t addr)
{
    return *(volatile const uint32_t *)addr;
}

Main Flash is memory-mapped, so a normal load is enough. Verify/OTA also treat payload as a const uint8_t * at the slot address for CRC and SHA-256.

9. Who calls this

  • ota.c — unlock → erase inactive pages → write DATA chunks → rewrite the 1 KB config page → lock.
  • OTA flag / config — erase/program the page at OTA_CONFIG_ADDR (0x0800FC00).
  • SystemClock_Config — latency + prefetch only (no unlock needed).
